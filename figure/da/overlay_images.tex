% Danish labels placed over raster images with embedded English text.
% Coordinates are given in pixels of the source image, measured from the top left corner.
\newsavebox{\DaImgBox}
\newcommand{\DaPixelOverlay}[5]{% #1 graphics options, #2 file, #3 width px, #4 height px, #5 tikz code
  \sbox{\DaImgBox}{\includegraphics[#1]{#2}}%
  \def\DaW{#3}\def\DaH{#4}%
  \begin{tikzpicture}[x=\wd\DaImgBox, y=\ht\DaImgBox]
    \node[anchor=south west, inner sep=0] at (0,0) {\usebox{\DaImgBox}};
    #5
  \end{tikzpicture}}
\newcommand{\DaPx}[2]{({(#1)/\DaW},{1-(#2)/\DaH})}

  }}
