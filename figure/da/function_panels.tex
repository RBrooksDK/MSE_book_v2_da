% Danish redrawings of the function-family figures in Chapter 1.

\pgfplotsset{
  dapanel/.style={
    axis lines=middle, width=5.6cm, height=5.6cm,
    xlabel={$x$}, ylabel={$y$},
    xlabel style={anchor=west}, ylabel style={anchor=south},
    tick label style={font=\scriptsize}, samples=200,
    every axis plot/.append style={very thick, color=mseViaBlue},
    clip=false
  }
}
\tikzset{
  danote/.style={font=\scriptsize, align=center, fill=white, inner sep=1pt},
  dapin/.style={-{Stealth[length=4pt]}, thin, black!70}
}
